7: def _sel_lon_lat(target, source):
8:     """Select a lon/lat slice of 'target' covering the full footprint of 'source' cells.
9: 
10:     The source grid uses cell CENTERS as coordinates.  Without correction the slice
11:     would stop at the outermost cell center, cutting off the outer half-cell on every
12:     edge (~0.025° for GloFAS v4).  We expand lo/hi by half the source resolution so
13:     the full cell footprint — center ± res/2 — is included in the JRC subset.
14:     """
15:     bounds = {}
16:     for coord in ['longitude', 'latitude']:
17:         lo, hi = _coord_bounds(source, coord)
18:         # Half-cell expansion: include the full footprint of boundary cells
19:         vals = source[coord].values
20:         res = float(np.median(np.abs(np.diff(vals)))) if len(vals) > 1 else 0.05
21:         lo -= res / 2
22:         hi += res / 2
23:         tv = target[coord].values; desc = tv[0] > tv[-1]
24:         bounds[coord] = slice(hi, lo) if desc else slice(lo, hi)
25:     return target.sel(bounds)

117:     with rasterio.open(out, 'w', **meta) as dst:
